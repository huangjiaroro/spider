ADD COLUMN `metric`  int(1) NULL DEFAULT 0 COMMENT '是否开启监控 0：否 1 是',
ADD COLUMN `metric_day_type`  int(1) NULL DEFAULT 0 COMMENT '监控日类型：0 全部 1：交易日',
ADD COLUMN `metric_start`  varchar(5) NULL COMMENT '监控开始时间时分 格式：hh:mm',
ADD COLUMN `metric_end`  varchar(5) NULL COMMENT '监控结束时间时分 格式：hh:mm',
ADD COLUMN `metric_interval`  int NULL DEFAULT 0 COMMENT '监控执行间隔',
ADD COLUMN `interval_type`  int NULL DEFAULT 0 COMMENT '时间间隔时间类型： 0：秒 1：分钟 2：小时',
ADD COLUMN `alert_threshold`  int(2) NULL DEFAULT 1 COMMENT '告警阈值',
ADD COLUMN `arrival_rate`  varchar(255) NULL COMMENT '到达率告警阈值',
ADD COLUMN `earlier_rate`  varchar(255) NULL COMMENT '同比率告警阈值';


ALTER TABLE spider.`spider_data_amount_summary`

CREATE TABLE spider.`alert_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `alert_content` varchar(255) DEFAULT NULL COMMENT '告警内容',
  `alert_level` int(1) DEFAULT '1' COMMENT '告警级别',
  `alert_person` varchar(100) DEFAULT NULL COMMENT '告警人',
  `alert_state` int(1) DEFAULT '0' COMMENT '告警状态 0：未告警 1：告警中 2：告警成功 3：告警失败',
  `alert_retry` int(11) DEFAULT '0' COMMENT '重试次数',
  `error_reason` text COMMENT '告警失败信息',
  `create_time` timestamp NULL DEFAULT NULL COMMENT '创建时间',
  `alert_time` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;


CREATE TABLE spider.`channel_node` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `node_name` varchar(100) DEFAULT NULL COMMENT '节点名称',
  `node_type` varchar(10) DEFAULT NULL COMMENT '节点类型',
  `alert` int(1) DEFAULT '0' COMMENT '是否进行节点监控 0：否 1：是',
  `metric_element` text COMMENT '监控环境参数配置Json',
  `alert_person` varchar(255) DEFAULT NULL COMMENT '告警人 多个逗号分隔',
  `alert_level` int(1) DEFAULT NULL COMMENT '告警级别',
  `metric_info` text COMMENT '节点信息Json',
  `work_tree` varchar(100) DEFAULT NULL COMMENT '业务树',
  `parent_id` varchar(11) DEFAULT NULL COMMENT '上一节点id',
  `node_id` varchar(11) NOT NULL,
  `channel_id` int(11) DEFAULT NULL COMMENT '通道Id',
  `draw_json` varchar(255) DEFAULT NULL COMMENT '点位置信息JSON',
  PRIMARY KEY (`id`),
  UNIQUE KEY `idx_channel_node_id` (`node_id`) USING BTREE,
  KEY `idx_channel_node_parent_id` (`parent_id`) USING BTREE,
  KEY `idx_channel_node_channel_id` (`channel_id`) USING BTREE
) ENGINE=InnoDB AUTO_INCREMENT=12 DEFAULT CHARSET=utf8;
